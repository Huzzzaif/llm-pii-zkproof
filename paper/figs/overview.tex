% =====================================================================
% Fig. 1 — ZKPriv overview (thesis-figure-skill, mode C, medium + hero)
% Core claim:  the provider can only produce a receipt if every in-circuit
%              automaton scans the committed response without reaching an
%              accepting state; reaching one (PII) leaves no proof.
% Archetype:   protocol flow with one central hero.
% Evidence:    hero = white prover box with a byte strip feeding a state
%              chain that either never accepts (green, => proof) or reaches
%              the accepting state a_k (red, => no proof).
% Can it be less?  Commitment / verifier-key boxes and metrics panel removed.
% Layout (form B), width <= 8.5 cm: provider zone right (LLM, optional
% tier-2 enclave, hero), user left, auditor + registry strip below.
% Revision 3 (after 4-lens review): enclave input y; receipt r defined in
%   caption; neutral byte text; accept/reject branch in hero; step markers
%   on the outer (left) side of their own arrow; no label crosses the zone
%   border; LLM/enclave edges aligned with the hero; white hero; smaller
%   secondary text; provider label inside the zone; tighter bottom row.
%
% Included by paper.tex via \input{figs/overview}. Needs TikZ libraries:
% arrows.meta, bending, positioning, fit, backgrounds, calc.
% =====================================================================

\definecolor{regFill}{HTML}{F2F1EE}  \definecolor{regLine}{HTML}{6B6A64}
\definecolor{okGreen}{HTML}{1B8F63}  \definecolor{badRed}{HTML}{C0392B}
\definecolor{ink}{HTML}{1F1F1E}
\pgfdeclarelayer{zone}\pgfdeclarelayer{panel}
\pgfsetlayers{zone,panel,main}
\begin{tikzpicture}[
  font=\footnotesize, text=ink,
  box/.style={rectangle, rounded corners=2.5pt, align=center, inner sep=3pt,

  enc/.style={box, fill=encBg, draw=encLine, densely dashed},
  reg/.style={box, fill=regFill, draw=regLine},
  arrow/.style={-{Stealth[length=4pt 1.3, width'=0pt 0.6]}, line width=0.8pt,

  lbl/.style={font=\scriptsize},

% ---------------- hero: prover with accept/reject automaton story ----------------
\node[font=\footnotesize\bfseries, anchor=west] (htitle) at (0,0) {ZKPriv prover (in ZK)};
\foreach \c [count=\i from 0] in {Y,o,u,r,{},o,r,d,e,r}{
  \node[byte] (b\i) at ($(htitle.south west)+(0.15+\i*0.30, -0.34)$) {\c};
}
\node[anchor=west, font=\scriptsize] at ($(b9.east)+(0.02,0)$) {\dots};
\foreach \k in {2,1}{\draw[provLine!25, line width=0.5pt, rounded corners=1.2mm]
        ($(b0)+(-0.17+0.05*\k,-0.60+0.05*\k)$) rectangle ($(b8)+(0.17+0.05*\k,-0.32+0.05*\k)$);}
\foreach \i in {0,2,4,6,8}{ \node[st] (s\i) at ($(b\i)+(0,-0.46)$) {}; }
\draw[-{Stealth[length=2pt]}, line width=0.45pt, color=provLine] ($(s0.west)+(-0.16,0)$) -- (s0.west);
\node[font=\tiny, text=provLine!80, anchor=north east] at ($(b8)+(0.27,-0.66)$) {$\times 13$};
\foreach \i/\j in {0/2,2/4,4/6,6/8}{
  \draw[-{Stealth[length=2pt]}, line width=0.45pt, color=provLine] (s\i) -- (s\j);
}
\node[anchor=west, font=\scriptsize, text=okGreen] (ok) at ($(s8.east)+(0.30,0.12)$)
      {no accept $\Rightarrow$ $\pi$ \checkmark};
\draw[-{Stealth[length=2pt]}, line width=0.45pt, color=okGreen] (s8) -- (ok.west);
\node[circle, draw=badRed, double, double distance=0.6pt, minimum size=2.6mm, inner sep=0pt,
      fill=badRed!12] (ak) at ($(s8)+(0.55,-0.42)$) {};
\node[anchor=west, font=\scriptsize, text=badRed] (bad) at ($(ak.east)+(0.06,0)$)
      {$a_k$ (PII) $\Rightarrow$ no $\pi$};
\draw[-{Stealth[length=2pt]}, line width=0.45pt, color=badRed, densely dashed] (s8) -- (ak);
\node[anchor=north west, font=\scriptsize, align=left, text=ink!85] (hfoot)
      at ($(b0.west |- ak.south)+(0,-0.06)$)
      {13 automata scan every byte; also binds\\
       length, padding, $\mathit{com}$, policy hash $h_{\mathrm{pol}}$};
\begin{pgfonlayer}{panel}
  \node[draw=provLine, line width=1.1pt, fill=white, rounded corners=3pt,

% ---------------- provider zone: LLM + optional enclave above the hero ----------------
